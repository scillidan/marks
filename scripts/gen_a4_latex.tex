% Inline code stays close to the listings look.
\newcommand{\postcodespan}[1]{%
  \textcolor{black}{\texttt{#1}}%
}

% Fenced code blocks are rendered with listings.
% Only #1 (the temp file written by the markdown package) is consumed. Older
% markdown releases pass 2 args to this renderer, newer ones pass 3; never
% reference #3 here or compilation breaks on TeX Live 2023.
\newcommand{\postinputfencedcode}[2]{%
  \lstinputlisting[style=postcode]{#1}%
}

% Notes are appended to \postnotes@acc in the SAME compile run and printed at
% the end by \postprintendnotes. This avoids the `endnotes` package, which ships
% markdown-rendered note content (links, %, #, ...) through an aux file whose
% line breaks / catcode changes break URLs. \long is required because note
% bodies may span multiple paragraphs (\par).
\newcounter{postendnotecounter}
\makeatletter
\def\postnotes@acc{}
\long\def\postendnote#1{%
  \stepcounter{postendnotecounter}%
  \textsuperscript{\thepostendnotecounter}%
  \expandafter\g@addto@macro\expandafter\postnotes@acc{%
    \par\noindent{\normalfont\bfseries\thepostendnotecounter.}\ #1}%
}
\newcommand{\postprintendnotes}{%
  \ifnum\value{postendnotecounter}>0
    \section*{Notes}%
    \par
    \postnotes@acc
  \fi
}
\makeatother

% Typography: a compact, minimal scale tuned for double-column work.
% Base size comes from `responsive`; headings/captions are small but readable.
% Override here if you want a different feel.
\newcommand{\postheadingone}[1]{{\LARGE #1}\par}
\newcommand{\postheadingtwo}[1]{{\large #1}\par}
\newcommand{\postheadingthree}[1]{{\normalsize\bfseries #1}\par}

% Fitbox defaults: scale images to fit the available column/page width,
% and shrink them further if they would not fit vertically.
% The minimum constraints are relative to the column so they never exceed the
% max width/height for high-resolution images (whose natural size is huge).
\fitboxset{
  maxwidth=0.9\linewidth,
  maxheight=0.75\textheight,
  belowboxspace=0pt,
  minwidth=0.2\linewidth,
  minheight=0.2\linewidth,
}

% Shared vertical skip around images/figures.
\newlength{\postimageskip}
\setlength{\postimageskip}{0.4\baselineskip}

% Per-image height cap. Inside a group we shrink images so the group never
% becomes taller than one column.
\newlength{\postimageheight}
\newif\ifpostgroupmode
\postgroupmodefalse

% Space to reserve below a fitbox image for its caption, measured from the
% caption text so fitbox can shrink the image to leave room for both.
\newlength{\postcaptionheight}
\newsavebox{\postcaptionbox}

% Print a narrow, centered caption block only when #1 is non-empty.
% The caption text itself stays left-aligned; only the block is centered.
\newcommand{\postcaption}[1]{%
  \if\relax\detokenize{#1}\relax
  \else
    \vspace{0.3\baselineskip}%
    \begin{minipage}{0.8\linewidth}%
      \raggedright
      \footnotesize
      #1%
    \end{minipage}%
    \par
  \fi
}

% Group of consecutive standalone images: an unbreakable minipage so the
% images always move together across a column/page boundary.
\newenvironment{postimagegroup}{%
  \par
  \addvspace{\postimageskip}%
  \noindent\begin{minipage}{\linewidth}%
  \postgroupmodetrue
  \centering
}{%
  \postgroupmodefalse
  \end{minipage}%
  \par
  \addvspace{\postimageskip}%
}

% Markdown renderers ---------------------------------------------------------
% Image renderer receives: #1=alt(caption), #2=escaped src, #3=raw src.
%
% Links are built from the *raw* URL (#3), with every TeX-special character
% escaped to the control sequence hyperref understands (\# \% \_ ...). Doing
% this at render time matters: footnote content travels through endnote aux
% files, where '#' (old hyperref: "Illegal parameter number in \Hy@tempa") and
% '%' would otherwise be re-interpreted with the wrong catcode and break.
% Image paths below keep #3 because \includegraphics needs the real filename.
\ExplSyntaxOn
\tl_new:N \l_posthref_url_tl
\cs_new_protected:Npn \posthref #1 #2 {
  % Work on a detokenized copy so '#'/special tokens keep their catcode
  % regardless of whether the URL came from the markdown input (catcode 12),
  % an endnote aux file (catcode 6), or plain text.
  \tl_clear:N \l_posthref_url_tl
  \tl_map_inline:Nn { \tl_to_str:n { #2 } } {
    \str_case:nnF { ##1 } {
      { \c_hash_str }       { \tl_put_right:Nn \l_posthref_url_tl { \# } }
      { \c_percent_str }    { \tl_put_right:Nn \l_posthref_url_tl { \% } }
      { \c_ampersand_str }  { \tl_put_right:Nn \l_posthref_url_tl { \& } }
      { \c_underscore_str } { \tl_put_right:Nn \l_posthref_url_tl { \_ } }
      { \c_left_brace_str } { \tl_put_right:Nn \l_posthref_url_tl { \{ } }
      { \c_right_brace_str }{ \tl_put_right:Nn \l_posthref_url_tl { \} } }
      { ~ } { \tl_put_right:Nn \l_posthref_url_tl { \textasciitilde {} } }
      { \c_circumflex_str } { \tl_put_right:Nn \l_posthref_url_tl { \^{} } }
    } {
      \tl_put_right:Nn \l_posthref_url_tl { ##1 }
    }
  }
  \href { \l_posthref_url_tl } { #1 }
}
\ExplSyntaxOff

\markdownSetup{
  renderers = {
    image = {
      \par
      \addvspace{\postimageskip}
      % Keep the image and its caption on the same page/column.
      \noindent\begin{minipage}{\linewidth}
      \centering
      \ifpostgroupmode
        \setlength{\postimageheight}{0.4\textheight}%
      \else
        \setlength{\postimageheight}{0.75\textheight}%
      \fi
      \if\relax\detokenize{#1}\relax
        \setlength{\postcaptionheight}{0pt}%
      \else
        \sbox{\postcaptionbox}{%
          \begin{minipage}{0.8\linewidth}%
            \raggedright
            \footnotesize
            #1%
          \end{minipage}%
        }%
        \setlength{\postcaptionheight}{\dimexpr\ht\postcaptionbox
          +\dp\postcaptionbox+0.3\baselineskip\relax}%
      \fi
      \fitbox[
        maxwidth=0.9\linewidth,
        maxheight=\postimageheight,
        belowboxspace=\postcaptionheight,
      ]{%
        \includegraphics{\postimagedir#3}%
      }%
      \par
      \postcaption{#1}%
      \end{minipage}%
      \par
      \addvspace{\postimageskip}
      \leavevmode
    },
    link = {\posthref{#1}{#3}},
    emphasis = {{\itshape #1}},
    strongEmphasis = {{\bfseries #1}},
    headingOne = {\postheadingone{#1}},
    headingTwo = {\postheadingtwo{#1}},
    headingThree = {\postheadingthree{#1}},
    blockQuoteBegin = {\begingroup\itshape},
    blockQuoteEnd = {\endgroup\par},
    inputFencedCode = {\postinputfencedcode{#1}{#2}},
    codeSpan = {\postcodespan{#1}},
    note = {\postendnote{#1}},
  }
}

  \catcode`\_=12\relax
